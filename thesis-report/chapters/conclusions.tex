\chapter{نتیجه‌گیری و کارهای آینده}
\label{ch:conclusions}

\section{جمع‌بندی}
\label{sec:summary}

هدف این پروژه، طراحی، پیاده‌سازی و ارزیابی یک نمونه اولیه گفتار‌به‌گفتار
تمام‌دوطرفه برای زبان فارسی بود. دو مسیر به‌طور موازی پیش برده شد: تطبیق یک
مدل سرتاسری گفتار‌به‌گفتار، و یک سامانه آبشاری ماژولار با کنترل تمام‌دوطرفه.
نتیجه نامتقارن بود.

\مهم{سامانه آبشاری جریانی} به نمونه‌ای کارکردی رسید که با میکروفون باز کار
می‌کند: پایان گفتار را خودکار و پیش‌دستانه تشخیص می‌دهد، پاسخ را جریانی و
بند‌به‌بند تولید و پخش می‌کند، و در هنگام پخش به وقفه کاربر واکنش نشان
می‌دهد. در آزمون حلقه‌ای با مرورگر واقعی، میانگین تأخیر پایان گفتار تا نخستین
صدای دستیار \N{372} و صدک نود آن \N{431} میلی‌ثانیه بود و هیچ‌یک از \N{201}
نوبت بی‌پاسخ نماند یا از \N{500} میلی‌ثانیه نگذشت؛ بخش گرافیکی کل سامانه
\N{11.1}~\en{GiB} حافظه گرفت.
آزمون معنایی نهایی \N{40} ردیفی پاسخ‌گوی \en{Qwen3-4B} از هر هفت شرط خودکار
گذشت (مرتبط‌بودن \N{3.150} در برابر \N{1.425} خط پایه).

\مهم{تشخیص وقفه} با یک کنترل‌گر دومرحله‌ای به هدف دقت رسید: یک قاب گفتار
بلندتر از پژواک مورد انتظار صدای دستیار را کم می‌کند و سپس بازشناس گفتار، پس
از کنارگذاشتن واژه‌های شنونده، واژه‌های دستوری و پژواک خود دستیار، تصمیم
توقف یا ادامه را می‌گیرد. در شبیه‌سازی موقعیت استقرار از ضبط‌های واقعی، دقت
\N{0.933} و \en{F1} \N{0.880} با میانه \N{120} میلی‌ثانیه تا کم‌کردن صدا، و
در مرورگر واقعی دقت \N{0.825} و \en{F1} \N{0.817} به دست آمد. طبقه‌بند صوتی
انرژی/زیر‌و‌بمی/\en{MFCC} به‌تنهایی روی رخدادهای واقعی پادکست به \N{0.574}
محدود ماند؛ این نتیجه منفی دلیل طراحی واژگانی است.

\مهم{اجزا} روی پنل‌های تازه و با پروتکل پیش‌ثبت‌شده دوباره برگزیده شدند:
بازشناس \en{NeMo} دقیق‌ترین نامزد درون بودجه تأخیر روی رونوشت انسانی بود،
صدای \en{Mana} فهم‌پذیرترین صدا نزد دو شنونده خودکار (در برابر دیگر صداهای
\en{Piper}، \en{ParsVoice-XTTS} و مدل \en{VITS} آزمایشگاه، حتی پس از بهینه‌سازی
استنتاج آن)، و
هیچ پاسخ‌گوی جایگزینی به آستانه جایگزینی نرسید.

\مهم{مسیر مستقیم} به مدل قابل استقرار نرسید، اما علت اصلی آن یافته شد:
توکن‌ساز انگلیسی \en{Moshi} برای هر واژه فارسی \N{6.6} نشانه می‌خواهد و
\N{98.5} درصد نوبت‌ها از بودجه یک نشانه در قاب فراتر می‌روند. برداشتن این
گلوگاه با لاتین‌نویسی واجی بهترین نتیجه تولید را دو برابر کرد (\N{6/9} در برابر
\N{3/9}) و خروجی ساکت را به گفتار شنیدنی تبدیل کرد، اما گفتار هنوز فهم‌پذیر
نبود. تجربه \en{J-Moshi}~\cite{ohashi2025jmoshi} همین تشخیص را تأیید می‌کند:
انتقال \en{Moshi} به ژاپنی نیز با جایگزینی توکن‌ساز آغاز شد، اما پس از آن به
حدود \N{60,000} ساعت پیش‌آموزش گفت‌وگوی ژاپنی نیاز داشت، بیش از پانصد برابر
\N{108.6} ساعت داده این پروژه. مدل ورودی گفتاری \en{Gemma~4}، حتی با آداپتر خودتقطیری، از سامانه آبشاری
ضعیف‌تر بود.

\مهم{مجموعه داده} \N{6,754} جفت «گفتار کاربر و پاسخ» با \N{108.584} ساعت
صوت ممیزی‌شده و نشت گروهی صفر است. برچسب‌های آن خودکارند و برچسب انسانی
ندارند.

\section{پاسخ به پرسش پژوهش}
\label{sec:answer}

پرسش عملی این پروژه دو بخش داشت. بخش نخست: با بودجه یک پروژه کارشناسی، تا
چه اندازه می‌توان یک سامانه گفتار‌به‌گفتار فارسی تمام‌دوطرفه ساخت؟ پاسخ این
است که یک سامانه آبشاری جریانی با کنترل وقفه، معیارهای عددی پیشنهاده برای
تأخیر (میانگین و صدک نود زیر \N{500} میلی‌ثانیه) و تشخیص وقفه (بالای \N{80}
درصد) را در آزمون‌های خودکار برآورده می‌کند و در حافظه یک کارت \N{12}
گیگابایتی جا می‌شود؛ اما یک مدل سرتاسری فارسی، با معماری‌ها و توکن‌سازهای
موجود و در این بودجه، قابل دستیابی نبود.

بخش دوم: چه چیزی را می‌توان با شواهد قابل ممیزی اثبات کرد؟ پاسخ در
\mbox{جدول~\ref{tab:req-status}} آمده است. ن۱ و ن۴ برآورده‌اند؛ ن۲، ن۳ و ن۵
با شواهد خودکار (آزمون حلقه‌ای مرورگر و شبیه‌سازی) برآورده‌اند، ن۳ با معیار
پیشنهاده و برای پنل اصلی حتی با خوانش هر‌نوبت، و ن۵ با روشی غیر از ویژگی‌های
صوتی تعیین‌شده؛
ن۶ یک انحراف اعلام‌شده است (حافظه سنجیده شد، سرعت روی سخت‌افزار هدف نه)؛ و
ن۷ برآورده نشد، چون مطالعه شنیداری انجام نشد. سهم روشمند نوشتار در همین
تفکیک است: برای هر سنجه، مرز ادعای مجاز مشخص شده و نتایج منفی همان‌قدر
گزارش شده‌اند که نتایج مثبت.

\section{پاسخ به پرسش‌های پژوهش پیشنهاده}
\label{sec:rq-answers}

\مهم{پرسش ۱: تطبیق کم‌هزینه یک مدل گفتار‌به‌گفتار موجود.} تطبیق کم‌پارامتر
\en{Moshika} با آداپتر \en{LoRA} روی یک \en{H100} هدف آموزشی را بهبود داد:
روی آزمون کنارگذاشته، سرگشتگی نشانه‌های متنی از \N{2.53} به \N{1.19} و
نشانه‌های صوتی از \N{13.98} به \N{4.71} رسید. اما در هیچ‌یک از هفت نسل،
گفتار فارسی فهم‌پذیر تولید نشد. علت اصلی در توکن‌ساز است: توکن‌ساز انگلیسی
برای هر واژه فارسی \N{6.6} نشانه می‌خواهد، در حالی که مدل در هر قاب
\N{12.5} هرتزی حداکثر یک نشانه متنی می‌سازد. لاتین‌نویسی واجی این بار را به
\N{0.67} نشانه در قاب رساند و بهترین اجرا را دو برابر کرد (\N{6/9})، اما کافی
نبود. پاسخ: با «کمترین هزینه آموزش مجدد» نه؛ تطبیق این معماری به فارسی دست‌کم
یک نمایش متنی سازگار با فارسی و آموزشی فراتر از یک آداپتر کوچک می‌خواهد.

\مهم{پرسش ۲: تغییرات معماری لازم برای تمام‌دوطرفه زیر \N{500} میلی‌ثانیه.}
در مسیر آبشاری این تغییرات لازم و کافی بودند: (۱) تشخیص پیش‌دستانه پایان گفتار
که کار را پیش از قطعی‌شدن پایان آغاز می‌کند و صوت را تا تأیید نگه می‌دارد؛ (۲)
تولید جریانی توکن با قید «فقط خط فارسی» در زمان رمزگشایی؛ (۳) بندبندی که
نخستین تکه را با دو تا چهار واژه می‌سازد و هر بند را بی‌درنگ به گفتار
برمی‌گرداند؛ (۴) یک صف پخش در کارخواه و یک دروازه نگه‌داشت صوت در کارساز که
با هم نقش «بافر دوگانه» را دارند؛ و (۵) کنترل وقفه دومرحله‌ای: کم‌کردن صدا
در یک قاب و تأیید واژگانی با کنارگذاشتن واژه‌های شنونده و پژواک، به‌همراه
نگه‌داشتن فقط بخش شنیده‌شده پاسخ قطع‌شده در تاریخچه. با این طراحی، میانگین
تأخیر \N{372} و صدک نود \N{431} میلی‌ثانیه شد و پاسخ پس از وقفه به اندازه پاسخ
نوبت عادی مرتبط ماند (بخش~\ref{sec:res-latency}).
در مسیر مستقیم، تغییر لازم پیش از هر تغییر معماری، توکن‌ساز است.

\مهم{پرسش ۳: بده‌بستان تأخیر، اندازه و کیفیت.} در بازشناسی و گفتارسازی اندازه،
کیفیت نخرید (جدول~\ref{tab:tradeoff}). دو بازشناس کوچک‌تر تنظیم‌شده روی فارسی
(\N{115} و \N{117} میلیون پارامتر) از همه نامزدهای بزرگ‌تر جز یکی دقیق‌تر
بودند، از جمله یک مدل سه‌میلیارد پارامتری تنظیم‌شده روی فارسی، و تنها مدل
دقیق‌تر، که خود روی فارسی تنظیم شده بود، بیش از ده برابر کندتر بود.
کوچک‌ترین گفتارساز (\N{15.7} میلیون پارامتر) هم فهم‌پذیرترین و هم سریع‌ترین بود،
و بزرگ‌ترین مدل، یعنی مدل سرتاسری \N{7.7} میلیارد پارامتری، به گفتار فهم‌پذیر
نرسید. در پاسخ‌گو پاسخ به داور بستگی دارد. زیر داور \en{Qwen} و \en{Gemini}،
پاسخ‌گوهای \N{9} تا \N{36} میلیارد پارامتری افزایشی نسبت به بهترین مدل
چهارمیلیاردی نداشتند (جدول~\ref{tab:resp-size})؛ زیر دو داور از خانواده‌هایی بدون
نامزد، مقایسه سومی از \N{15} مدل و ساخت‌های کوانتیده آن‌ها، \en{Gemma~4 12B}
چهاربیتی را \N{1.2} تا \N{1.75} امتیاز (از \N{10}) بهتر از بهترین مدل
چهارمیلیاردی و \N{3.5} امتیاز بهتر از پاسخ‌گوی مستقر یافت، با نخستین توکن در
\N{39} میلی‌ثانیه و \N{10.9} گیگابایت حافظه (جدول~\ref{tab:resp-v3}). مدل‌های
بزرگ‌تر سود چندانی بیشتر نداشتند و هزینه تأخیر داشتند، و با تنظیمات سرویس‌دهی
مقایسه‌شده هیچ پاسخ‌گوی بهتری با دروازه تأخیر در کارت \N{12} گیگابایتی جا نشد
(تنظیمی در \en{llama.cpp} که بعدتر یافت شد \en{Gemma~4 12B} را در هر دو مرز جای داد).
اما در سامانه زنده این مدل
هنگامی که بازشناس جمله بعدی را مخدوش کرد از کاربر تکرار خواست، و پاسخ‌هایش پس
از وقفه‌های دیررس در هر سه پیکربندی آزموده به قاعده انسجام ازپیش‌ثبت‌شده نرسید؛ پس
صلاحیت استقرار نیافت.
بهترین نقطه بده‌بستان برای فارسی بی‌درنگ، بازشناس‌ها و گفتارسازهای کوچک و
تخصصی‌شده برای فارسی همراه با بزرگ‌ترین پاسخ‌گویی است که نیم‌رخ حافظه به‌صورت
چهاربیتی اجازه می‌دهد، به شرط آن‌که در سامانه زنده سنجیده شود، جایی که خطای
بازشناسی معنای پاسخ خوب را تغییر می‌دهد.

\مهم{پرسش ۴: درک سالمندان.} ابزار و پروتکل مطالعه شنیداری آماده و
ازپیش‌ثبت‌شده است، اما شرکت‌کننده‌ای در دسترس نبود و مطالعه انجام نشد؛ ازاین‌رو
این نوشتار درباره درک سالمندان ادعایی ندارد.

\section{کارهای آینده}
\label{sec:future}

\subsection{تکمیل شواهد}

\begin{enumerate}
\item \مهم{سخت‌افزار هدف و مسیر فیزیکی.} اجرای همان پروتکل حلقه‌ای روی یک
      کارت \N{12} تا \N{24} گیگابایتی، و یک آزمون با میکروفون و بلندگوی
      فیزیکی که پژواک واقعی و حذف پژواک مرورگر را در بر بگیرد.
\item \مهم{مطالعه انسانی.} اجرای مطالعه شنیداری ثبت‌شده با \N{5} تا \N{10}
      شرکت‌کننده (دست‌کم دو نفر بالای \N{60} سال)؛ و در گام بعد، یک جلسه کوتاه
      تعامل زنده با سالمندان.
\item \مهم{برچسب انسانی.} بازبینی شنیداری نمونه‌ای از رخدادهای وقفه و
      جفت‌های مجموعه داده توسط دو بازبین مستقل، تا برچسب‌های خودکار
      \en{Gemini} و قواعد استخراج اعتبارسنجی شوند.
\item \مهم{داوری انسانی پاسخ‌گو.} دو داور خودکار نخست درباره پاسخ‌گوها اختلاف
      زیادی داشتند (\en{Gemini} پاسخ‌گوهای با پرامپت سالمندان را حدود یک امتیاز
      بهتر می‌دانست و داور \en{Qwen} نه)؛ دو داور بعدی از خانواده‌های دیگر با
      یکدیگر هم‌نظرند و مدل‌های بزرگ‌تر \en{Gemma} را ترجیح می‌دهند. قاعده
      جایگزینی سامانه \N{12} گیگابایتی پیش از رسیدن مطالعه شنیداری به شرکت‌کنندگان ثبت شد و
      پیکربندی \en{Qwen3.5-4B} با پرامپت سالمندان آزمون استقرار آن را گذرانده
      است؛ تنها ترجیح شرکت‌کنندگان، که گردآوری نشد، برای جایگزینی آن کم است. این‌که انسان‌ها
      با ترجیح داوران خودکار برای پاسخ‌گوی \N{24} گیگابایتی هم‌نظرند یا نه، و
      این‌که سالمندان درخواست تکرار آن پس از خطای بازشناسی را می‌پذیرند یا نه،
      سنجیده نشده است.
\end{enumerate}

\subsection{بهبود سامانه}

\begin{enumerate}
\item \مهم{پاسخ‌گویی که می‌داند بازشناس خطا می‌کند.} پاسخ‌گوی \N{24} گیگابایتی پس
      از بیشتر جمله‌های مخدوش از کاربر تکرار خواست و پاسخ‌گوهای چهارمیلیاردی حدس
      زدند. دادن اطمینان بازشناس به پاسخ‌گو، یا دستور پاسخ به محتمل‌ترین درخواست
      همراه با تأییدی کوتاه، می‌تواند کیفیت مدل بزرگ‌تر را در سامانه زنده نگه دارد؛
      این کار آزمون ازپیش‌ثبت‌شده خود را لازم دارد.
\item \مهم{تأیید واژگانی دقیق‌تر.} واج نخست بازخوردهای کوتاه گاهی بریده می‌شود
      («آره» به «کار»). افزودن \N{200} میلی‌ثانیه صوت پیش از آغاز به ورودی
      بازشناس آزموده شد و رد شد: وقفه‌های بیشتری متوقف شدند (\N{0.98} در برابر
      \N{0.95}) اما بازخوردها و نویزهای بیشتری هم، و دقت کم شد
      (بخش~\ref{sec:res-bargein-sim}). جداکردن واژه شنونده از واژه محتوایی در
      آغاز بریده‌شده به داده برچسب‌دار و مدلی آموخته نیاز دارد.
\item \مهم{تشخیص پایان گفتار برای سالمندان.} درنگ‌های بلند گفتار آهسته گاهی
      پیش از پایان جمله نوبت را می‌بندد (\N{87} نوبت سرور برای \N{80} نوبت
      برنامه‌ریزی‌شده). سامانه این را با ادامه‌دادن کاربر و توقف پاسخ جبران
      می‌کند. دروازه معنایی پایان نوبت در دو دور آزموده شد (بخش~\ref{sec:res-latency}):
      بهترین آن، داوری کامل‌بودن با مدل زبانی، در آزمون حلقه‌ای گفته‌های کمتری را
      دوپاره کرد اما \N{15} درصد نوبت‌های تمام‌شده را نگه داشت و صدک نود را از
      \N{500} میلی‌ثانیه فراتر برد. گام بعدی دروازه‌ای با زمان نگه‌داشتن کوتاه‌تر، یا
      مدل پایان نوبت فارسی آموخته روی پایان‌های نوبت برچسب‌خورده به دست انسان است،
      نه مرز گوینده‌های پادکست.
\item \مهم{بازشناس دقیق‌تر در همان بودجه.} \en{Seamless} خطای نویسه کمتری
      داشت اما بیش از چهار برابر بودجه تأخیر؛ تقطیر آن در یک مدل \en{CTC}
      یا اجرای جریانی آن می‌تواند دقت را بدون هزینه تأخیر بالا ببرد.
\end{enumerate}

\subsection{مسیر مستقیم}

یافته توکن‌ساز مسیر روشنی پیش می‌گذارد: مدل مستقیم فارسی به توکن‌سازی نیاز
دارد که در بودجه یک نشانه در قاب جا شود (واج‌محور یا با واژگان فارسی)، به
پیش‌آموزش روی گفت‌وگوی فارسی در مقیاسی نزدیک به \en{J-Moshi} (ده‌ها هزار
ساعت، که پیکره عمومی آن برای فارسی وجود ندارد) و به راهکاری برای شکاف واداشت معلم و تولید آزاد، مانند حذف تصادفی زمان‌بندی‌شده
ورودی در مقیاس بزرگ‌تر یا آموزش با خروجی خود مدل. رویکرد مدل منجمد
\en{Freeze-Omni}~\cite{wang2025freezeomni} و تثبیت تولید گفتار فارسی با داده
تک‌گوینده فراوان مانند \en{ManaTTS}~\cite{qharabagh2025manatts} پیش از
آموزش گفت‌وگو نیز مسیرهای امیدبخشی‌اند. برای مدل ورودی گفتاری، آموزش روی گفتار
متنوع‌تر (و نه فقط پادکست) لازم است، چون خودتقطیری روی یک دامنه، بیرون از آن
دامنه آسیب زد.

\section{سخن آخر}
\label{sec:closing}

این پروژه نشان داد که یک دستیار گفتاری فارسی تمام‌دوطرفه با میکروفون باز،
تأخیر زیر نیم ثانیه و تشخیص وقفه قابل اتکا، در بودجه یک پروژه کارشناسی و با
اجزای متن‌باز قابل ساخت است؛ و نشان داد که مدل سرتاسری گفتار‌به‌گفتار فارسی به
چیزی بیش از تطبیق کم‌پارامتر نیاز دارد، از جمله توکن‌سازی متناسب با زبان.
نتیجه منفی مسیر مستقیم، همراه با علت سنجیده‌شده آن، ارزشی کمتر از نتیجه مثبت
مسیر آبشاری ندارد: کاهش زیان آفلاین می‌تواند گمراه‌کننده باشد و آزمون تولید
در زمان اجرا باید بخشی از معیار انتخاب باشد.
